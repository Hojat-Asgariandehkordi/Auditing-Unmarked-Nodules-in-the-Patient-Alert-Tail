%% Cross-references between the main paper and the supplement. main.tex builds
%% the paper with or without the supplement, and supplement.tex builds the
%% supplement alone, so a label may not exist in the current build.
%% \storedref{label}{value} is a live \ref when the label exists and prints the
%% stored value otherwise, so no "??" appears. If a stored value no longer
%% matches, the combined build warns ("Stored value ... is out of date").
%%   \mainref, \maineqref : supplement -> main paper (e.g. Eq. (1), Sec. 4.2)
%%   \suppsec             : main paper -> supplement ("Supp. Sec. D")
\usepackage{refcount}
\DeclareRobustCommand{\storedref}[2]{%
  \ifcsname r@#1\endcsname
    \ref{#1}%
    \edef\storedrefNow{\getrefnumber{#1}}\edef\storedrefStored{#2}%
    \ifx\storedrefNow\storedrefStored\else
      \PackageWarningNoLine{storedref}{Stored value #2 for '#1' is out of date: the document now has \storedrefNow}%
    \fi
  \else #2\fi}
\DeclareRobustCommand{\mainref}[2]{\storedref{#1}{#2}}
\DeclareRobustCommand{\maineqref}[2]{\textup{(\storedref{#1}{#2})}}
\DeclareRobustCommand{\suppsec}[2]{Supp.~Sec.~\storedref{#1}{#2}}
